func fat(n) {
    if (n <= 1) { return 1; }
    return n * fat(n - 1);
}
func fib(n) {
    if (n < 2) { return n; }
    return fib(n - 1) + fib(n - 2);
}
func mdc(a, b) {
    while (b != 0) { var r = a % b; a = b; b = r; }
    return a;
}
print("5! =", fat(5), " fib(15) =", fib(15), " mdc(48,18) =", mdc(48, 18));
