n = 5;
i = 0;
s = 0;
while (i < n) {         // i e s mudam no laço: não são propagados
    s = s + i * 1;
    i = i + 1;
}
t = s * 0 + n * 2;      // s*0 -> 0, n -> 5: t = 10
if (n > 10) print(999); else print(t);    // condição constante: só o else fica
while (0) print(1);                        // código morto
print(s - s + 2 * s);   // s - s -> 0;  2*s -> s+s
print(!!(i == n) && 1);
