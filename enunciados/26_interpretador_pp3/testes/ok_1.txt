# Fibonacci, primos e FizzBuzz
var a = 0;
var b = 1;
var n = 0;
while (n < 10) {
    print("fib(" + n + ") =", a);
    var t = a + b;
    a = b;
    b = t;
    n = n + 1;
}
var primos = "";
for i = 2 to 30 {
    var ehPrimo = true;
    var d = 2;
    while (d * d <= i and ehPrimo) {
        if (i % d == 0) { ehPrimo = false; }
        d = d + 1;
    }
    if (ehPrimo) { primos = primos + i + " "; }
}
print("primos até 30:", primos);
